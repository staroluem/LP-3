import heapq

def huffman_encoding(chars, freq):
    heap = [[weight, [char, ""]] for char, weight in zip(chars, freq)]
    heapq.heapify(heap)

    while len(heap) > 1:
        lo = heapq.heappop(heap)
        hi = heapq.heappop(heap)

        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]

        heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])

    # Extract final codes into dictionary
    huff_dict = sorted(heapq.heappop(heap)[1:], key=lambda p: chars.index(p[0]))
    return {char: code for char, code in huff_dict}


# Driver code
chars = ['a', 'b', 'c', 'd', 'e', 'f']
freq = [5, 9, 12, 13, 16, 45]

codes = huffman_encoding(chars, freq)

print("Huffman Codes:")
for ch, code in codes.items():
    print(f"{ch}: {code}")
